from fastapi import FastAPI, HTTPException from fastapi.testclient import TestClient from pydantic import BaseModel from typing import List, Optional from sqlalchemy import create_engine, Column, Integer, String, Float, MetaData, Table from sqlalchemy.orm import sessionmaker, declarative_base from contextlib import contextmanager # --- Pydantic Models --- class AxisScore(BaseModel): server_id: int axis_name: str score: float tier: str class ServerDetail(BaseModel): server_id: int hostname: str risk_scores: List[AxisScore] overall_risk: AxisScore recent_activity: List[str] # Placeholder for actual recent activity data total_servers: int # --- Database Setup (In-Memory SQLite for Testing) --- DATABASE_URL = "sqlite:///:memory:" engine = create_engine(DATABASE_URL) SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine) Base = declarative_base() # Define SQLAlchemy Table for mcp_llm_axis_scores metadata = MetaData() mcp_llm_axis_scores_table = Table( 'mcp_llm_axis_scores', metadata, Column('server_id', Integer, primary_key=True), Column('axis_name', String, primary_key=True), Column('score', Float), Column('tier', String) ) # Define SQLAlchemy Table for servers (to get hostname and total count) servers_table = Table( 'servers', metadata, Column('server_id', Integer, primary_key=True), Column('hostname', String) ) # Create tables metadata.create_all(engine) @contextmanager def get_db(): db = SessionLocal() try: yield db finally: db.close() # --- FastAPI Application --- app = FastAPI() # --- Helper Functions --- def get_risk_tiers(score: float) -> str: if score >= 0.8: return "Critical" elif score >= 0.5: return "High" elif score >= 0.2: return "Medium" else: return "Low" def seed_database(): with get_db() as db: # For total count ] for item in servers_data: db.execute( servers_table.insert().values(**item) ) db.commit() # --- API Endpoint --- @app.get("/servers/{server_id}/detail", response_model=ServerDetail) async def get_server_detail(server_id: int): with get_db() as db: # Fetch server hostname server_stmt = servers_table.select().where(servers_table.c.server_id == server_id) server_result = db.execute(server_stmt).fetchone() if not server_result: raise HTTPException(status_code=404, detail="Server not found") hostname = server_result.hostname # Fetch all axis scores for the server axis_scores_stmt = mcp_llm_axis_scores_table.select().where(mcp_llm_axis_scores_table.c.server_id == server_id) axis_scores_results = db.execute(axis_scores_stmt).fetchall() risk_scores = [] total_score = 0 for row in axis_scores_results: score_obj = AxisScore( server_id=row.server_id, axis_name=row.axis_name, score=row.score, tier=row.tier ) risk_scores.append(score_obj) total_score += row.score # Calculate overall risk (average of all axes) overall_score = total_score / len(axis_scores_results) if axis_scores_results else 0 overall_risk_tier = get_risk_tiers(overall_score) overall_risk_score = AxisScore( server_id=server_id, axis_name="overall_risk", score=round(overall_score, 2), tier=overall_risk_tier ) # Fetch total number of servers total_servers_stmt = servers_table.select() total_servers_count = db.execute(total_servers_stmt).rowcount # Placeholder for recent activity recent_activity = [f"Activity log entry for server {server_id} at {i}" for i in range(3)] return ServerDetail( server_id=server_id, hostname=hostname, risk_scores=risk_scores, overall_risk=overall_risk_score, recent_activity=recent_activity, total_servers=total_servers_count ) #
