import requests
from fastapi import FastAPI, HTTPException
from fastapi.responses import JSONResponse
import logging
import time
from typing import Dict, List

app = FastAPI()

# Configure logging setup
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Constants
write_service_url = "http://127.0.0.1:8772/write"
inference_router_url = "http://127.0.0.1:8773"

def run():
    logger.info("Starting ZO-SENTINEL")
    
    try:
        # Create table if not exists
        response = requests.post(write_service_url, json={'table': 'write_service', 'rows': {}, 'wait': True})
        if response.status_code != 201:
            raise Exception(f"Failed to create table write_service")

        # Insert data into write_service table
        response = requests.post(write_service_url, json={'table': 'write_service', 'rows': {'service': 'mcp', 'type': 'DuckDB'}, 'wait': True})
        if response.status_code != 201:
            raise Exception(f"Failed to insert data into write_service")

        # Insert data into service_health table
        response = requests.post("/write", json={'table':'service_health','rows':{'service':'mcp','last_heartbeat':time.time()}})
        if response.status_code != 201:
            raise Exception(f"Failed to insert data into service_health")

    except Exception as e:
        logger.error(f"Error: {e}")
    
    finally:
        # Delete table and reset write_service
        response = requests.post(write_service_url, json={'table': 'write_service', 'rows': {}, 'wait': True})
        if response.status_code != 200:
            raise Exception(f"Failed to delete table write_service")

@app.post("/write")
def send_data(data: Dict):
    try:
        # Insert data into service_health table
        response = requests.post("/write", json={'table':'service_health','rows':{'service':data['service'],'last_heartbeat':time.time()}})
        if response.status_code != 201:
            raise Exception(f"Failed to insert data into service_health")
    except Exception as e:
        logger.error(f"Error: {e}")
    
    return JSONResponse({"message": "Data sent successfully"}, status_code=200)

if __name__ == "__main__":
    try:
        run()
    except Exception as e:
        logger.error(f"Error: {e}")
